/* SONDA audytu 05-architekt (scratchpad, nie w repo): ARCH-05 — limit notatki serii: import CSV 500 znaków vs edycja 300 znaków. */
import * as store from '@/lib/store';
import { parseImport } from '@/lib/csvparse';
import { commitCsvImport } from '@/lib/csvimport';
import { clampName } from '@/lib/state/util';
import { IMPORT_NOTE_MAX } from '@/lib/csvcore';
import { fresh } from './helpers';

test('notatka serii 450 znaków z CSV zostaje w historii; zapis z okna „Notatka do serii” (clampName(…, 300)) ją ucina', async () => {
  await fresh();
  const long = 'x'.repeat(450);
  const csv = 'Date,Workout Name,Duration,Exercise Name,Set Order,Weight,Reps,Distance,Seconds,Notes,Workout Notes,RPE\n' +
    `2026-03-02 18:05:00,Push A,52m,Bench Press (Barbell),1,60.0,8.0,0,0,${long},,\n`;
  const file = parseImport(csv);
  const r = await commitCsvImport(file, {} as any);
  const w = store.getState().workouts.find(x => x.templateName === 'Push A')!;
  const set = w.exercises[0].sets[0];
  console.log(JSON.stringify({ IMPORT_NOTE_MAX, imported: r, noteBefore: set.note.length }));
  expect(set.note.length).toBe(450);
  /* ten sam kod co app/history/edit/[id].tsx:113 i components/WorkoutBlocks.tsx:134 po „Zapisz” bez zmiany tekstu */
  set.note = clampName((set.note ?? '').trim(), 300);
  console.log(JSON.stringify({ noteAfterEditSave: set.note.length }));
  expect(set.note.length).toBe(300);
});
