import * as store from '@/lib/store';
import { applyLang } from '@/lib/i18n';
import { PAYWALL, type Entitlement } from '@/lib/access';
import { setStoreAdapter, resetStoreAdapterForTests } from '@/lib/store-adapter';
import { createFakeStore } from '@/lib/store-fake';
import { resetEntitlementsForTests } from '@/lib/entitlement';
import { planName } from '@/app/paywall';
import { fresh, saved } from './helpers';
import { renderApp, flushAll, screen, act } from './app';

const NOW = new Date(2026, 9, 14, 12, 0).getTime(); const DAY = 86400e3;
const settle = async () => { for (let i = 0; i < 8; i++) await flushAll(10); };
afterEach(() => { PAYWALL.ENABLED = false; resetEntitlementsForTests(); resetStoreAdapterForTests(); applyLang('pl'); jest.useRealTimers(); });

test('SONDA: posiadacz rocznej subskrypcji — co widzi na ekranie zakupu z Ustawień', async () => {
  jest.useFakeTimers({ now: NOW }); await fresh(undefined, 'pl'); resetEntitlementsForTests(); PAYWALL.ENABLED = true;
  const yearly: Entitlement = { plan: 'yearly', source: 'purchase', expiresAt: NOW + 300 * DAY };
  setStoreAdapter(createFakeStore([yearly])); store.save(); await act(async () => { await store.flush(); });
  await renderApp({ saved: JSON.parse(JSON.stringify(saved())), locale: 'pl', url: '/paywall?reason=settings' }); jest.setSystemTime(NOW); await settle();

    kartaPosiadania: !!screen.queryByTestId('paywall-owned'),

  };

  console.log('SONDA-08', JSON.stringify(fakty));
  expect(fakty).toEqual({ kartaPosiadania: true, naglowek5DniZaDarmo: true, listaPlanow: ['plan-yearly', 'plan-monthly', 'plan-lifetime'], zaznaczonyPlan: ['plan-yearly'], przyciskSubskrybuj: true, nazwaPosiadanegoPlanuNaKarcie: false });
});
